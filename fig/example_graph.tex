% Small example directed graph used across all schema figures
% Nodes: 1, 2, 3, 4
% Edges: 1->2, 1->3, 2->3, 3->4
\begin{tikzpicture}[
    node distance=1.2cm,
    vertex/.style={circle, draw, thick, minimum size=5.5mm, inner sep=0pt, font=\footnotesize\bfseries},
    edge/.style={-{Stealth[length=2mm]}, thick},
  ]
  \node[vertex] (1) {1};
  \node[vertex, right of=1] (2) {2};
  \node[vertex, below of=2] (3) {3};
  \node[vertex, right of=3] (4) {4};

  \draw[edge] (1) -- (2);
  \draw[edge] (1) -- (3);
  \draw[edge] (2) -- (3);
  \draw[edge] (3) -- (4);
\end{tikzpicture}
