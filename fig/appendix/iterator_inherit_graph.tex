\documentclass[tikz,border=3mm]{standalone}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usetikzlibrary{arrows.meta}

\begin{document}
\begin{tikzpicture}[
  base/.style={font=\small\ttfamily\bfseries},
  abst/.style={font=\small\ttfamily, anchor=west},
  impl/.style={font=\small\ttfamily, text=black!70, anchor=west},
  inh/.style={-{Triangle[open,length=1.8mm,width=1.4mm]}, semithick},
  ext/.style={-, thin, black!50},
]

\def\LV{5.5mm}   % vertical spacing between child lines
\def\I{4mm}       % indent for concrete classes
\def\CS{42mm}     % column separation

% ── Root ──
\node[base] (root) at ({1.5*\CS}, 0) {table\_iterator};

% ── Abstract cursors ──
\node[abst] (out) at (0, -12mm) {OutCursor};
\node[abst] (inc) at (\CS, -12mm) {InCursor};
\node[abst] (nod) at ({2*\CS}, -12mm) {NodeCursor};
\node[abst] (edg) at ({3*\CS}, -12mm) {EdgeCursor};

% ── Root to abstract cursors (comb) ──
\coordinate (branch) at ({1.5*\CS}, -5mm);
\draw[inh] (branch) -- (root.south);
\draw[ext] (0, -5mm) -- ({3*\CS}, -5mm);
\draw[ext] (0, -5mm) -- (out.north west);
\draw[ext] (\CS, -5mm) -- (inc.north west);
\draw[ext] ({2*\CS}, -5mm) -- (nod.north west);
\draw[ext] ({3*\CS}, -5mm) -- (edg.north west);

% ── OutCursor children ──
\node[impl] (a1) at (\I, {-12mm-\LV}) {AdjOutCursor};
\node[impl] (a2) at (\I, {-12mm-2*\LV}) {EkeyOutCursor};
\node[impl] (a3) at (\I, {-12mm-3*\LV}) {SplitEKeyOutCursor};

\draw[ext] ([xshift=-1.5mm]a1.west) -- ([xshift=-1.5mm]a4.west);
\foreach \n in {a1,a2,a3,a4}
  \draw[ext] ([xshift=-1.5mm]\n.west) -- (\n.west);
\draw[inh] ([xshift=-1.5mm]a1.west) -- ([yshift=-0.5mm]out.south west);

% ── InCursor children ──
\node[impl] (b1) at ({\CS+\I}, {-12mm-\LV}) {AdjInCursor};
\node[impl] (b2) at ({\CS+\I}, {-12mm-2*\LV}) {EkeyInCursor};
\node[impl] (b3) at ({\CS+\I}, {-12mm-3*\LV}) {SplitEkeyInCursor};
\node[impl] (b4) at ({\CS+\I}, {-12mm-4*\LV}) {StdInCursor};

\draw[ext] ([xshift=-1.5mm]b1.west) -- ([xshift=-1.5mm]b4.west);
\foreach \n in {b1,b2,b3,b4}
  \draw[ext] ([xshift=-1.5mm]\n.west) -- (\n.west);
\draw[inh] ([xshift=-1.5mm]b1.west) -- ([yshift=-0.5mm]inc.south west);

% ── NodeCursor children ──
\node[impl] (c1) at ({2*\CS+\I}, {-12mm-\LV}) {AdjNodeCursor};
\node[impl] (c2) at ({2*\CS+\I}, {-12mm-2*\LV}) {EkeyNodeCursor};
\node[impl] (c3) at ({2*\CS+\I}, {-12mm-3*\LV}) {SplitEKeyNodeCursor};
\node[impl] (c4) at ({2*\CS+\I}, {-12mm-4*\LV}) {StdNodeCursor};

\draw[ext] ([xshift=-1.5mm]c1.west) -- ([xshift=-1.5mm]c4.west);
\foreach \n in {c1,c2,c3,c4}

\node[impl] (d2) at ({3*\CS+\I}, {-12mm-2*\LV}) {EkeyEdgeCursor};
