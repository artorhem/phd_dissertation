      class table_iterator
      {
      protected:
        WT_CURSOR *cursor = nullptr;
        WT_SESSION *session = nullptr;
        bool is_first = true;
        bool has_next = true;
      
      public:
      table_iterator() = default;
      virtual ~table_iterator() = default;
      [[nodiscard]] bool has_more() const { return has_next; };
      virtual void reset()
      {
          cursor->reset(cursor);
          is_first = true;
          has_next = true;
      }
      void close() { cursor->close(cursor); }
      };
    \end{lstlisting}
